import MovieCard from "./MovieCard";
import type { Movie } from "../types";

type MovieListProps = {
  movies: Movie[];
  viewMode: "grid" | "list";
};

function MovieList({ movies, viewMode }: MovieListProps) {
  if (movies.length === 0) {
    return <p>No movies found.</p>;
  }

  return (
    <div className={`movie-grid ${viewMode === "list" ? "list-view" : ""}`}>
      {movies.map((movie) => (
        <MovieCard
          key={movie.id}
          movie={movie}
          onClick={() => console.log(movie.title)}
        />
      ))}
    </div>
  );
}

export default MovieList;
